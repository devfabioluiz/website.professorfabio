import { useState } from 'react';

const Login = () => {
  const [formData, setFormData] = useState({
    email: '',
    senha: '',
  });

  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData({ ...formData, [name]: value });
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    console.log(formData);

  return (

      <h1>Login</h1>
      <div>
        <label>Email:</label>
        <input
          type="email"
          name="email"
          value={formData.email}
          onChange={handleChange}
        />
      </div>
      <div>
        <label>Senha:</label>
        <input
          type="password"
          name="senha"
          value={formData.senha}
          onChange={handleChange}
        />
      </div>
      <button type="submit">Entrar</button>
    </form>
  );
};

export default Login;

🤓 Explicando o código 📝

O estado formData armazena os campos email e senha. O handleChange usa dynamic property assignment ([name]: value) para atualizar apenas o campo digitado. O handleSubmit previne o recarregamento da página e, por enquanto, exibe os dados no console.

7ª Parte — Chamar a API no componente HOME

Conforme aprendemos na Aula 8, faça uma requisição GET com Axios para a URL https://projeto-node-step-t5i1.vercel.app/produtos no componente Home.jsx e exiba os nomes dos produtos em uma lista na tela.

Gabarito: 

import { useState, useEffect } from 'react';
import axios from 'axios';

const Home = () => {
  const [produtos, setProdutos] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    axios
      .get('https://projeto-node-step-t5i1.vercel.app/produtos')
      .then((response) => {
        setProdutos(response.data);
        setLoading(false);
      })
      .catch((err) => {
        setError(err.message);
        setLoading(false);
      });
  }, []);

  if (loading) return <p>Carregando produtos...</p>;
  if (error) return <p>Erro: {error}</p>;

  return (
    <div style={{ maxWidth: '600px', margin: '40px auto', padding: '0 20px' }}>
      <h1>Home</h1>
      <h2>Produtos</h2>
      <ul>
        {produtos.map((produto) => (
          <li key={produto._id}>{produto.nome}</li>
        ))}
      </ul>
    </div>
  );
};

export default Home;

🤓 Explicando o código 📝

O useEffect com array vazio executa a requisição apenas uma vez. axios.get() busca os dados, que ficam em response.data. Enquanto a resposta não chega, exibimos "Carregando produtos..."; se houver erro, mostramos a mensagem. Os nomes são renderizados com map.

9ª Parte — Criar o componente CriarProduto

Crie o componente CriarProduto.jsx na subpasta aulas-integradas com um formulário contendo todos os campos de um produto da API:

  • nome
  • preco
  • descricao
  • categoria
  • imagem
  • avaliacao, que possui nota e quantidade

Obs: os campos _id e __v são gerados automaticamente pela API, então não precisam ser preenchidos no formulário. Não se esqueça também de criar uma rota para esse novo componente no App.jsx.

🔐 Por que precisamos de um token?

A API protege as operações de criar, editar e deletar produto. Se tentarmos um POST, PUT ou DELETE em /produtos sem nos identificarmos, ela responde com erro 401 - Token não informado. Só a listagem (GET) é pública.

O token é como uma "carteirinha de acesso" (JWT) que a API devolve ao fazermos login ou registro. No POST /registrar (e no POST /login), a resposta já vem com o token em response.data.token.

Como armazenar o token: guardamos no localStorage, que sobrevive ao recarregar ou fechar o navegador: 


localStorage.setItem('token', response.data.token); // salva
const token = localStorage.getItem('token');        // lê depois

Como enviar o token: em toda requisição protegida, mandamos o header Authorization no formato Bearer <token>: 


axios.post(url, formData, {
  headers: { Authorization: `Bearer ${localStorage.getItem('token')}` },
});

Regra de ouro: GET é aberto; POST, PUT e DELETE de produto exigem o token no header. Vamos aplicar isso nas partes 10ª, 13ª e 14ª.

import { useState } from 'react';

const CriarProduto = () => {
  const [formData, setFormData] = useState({
    nome: '',

    descricao: '',
    categoria: '',

      <h1>Criar Produto</h1>

          name="nome"

        />

        <label>Preço:</label>
        <input
          type="number"
          step="0.01"
          name="preco"
          value={formData.preco}
          onChange={handleChange}
        />
      </div>
      <div>
        <label>Descrição:</label>

          value={formData.descricao}

          value={formData.categoria}
          onChange={handleChange}
        />
      </div>
      <div>
        <label>Imagem (URL):</label>
        <input
          type="text"
          name="imagem"
          value={formData.imagem}
          onChange={handleChange}
        />
      </div>
      <div>
        <label>Nota da avaliação:</label>
        <input
          type="number"
          step="0.1"
          name="nota"
          value={formData.avaliacao.nota}
          onChange={handleAvaliacaoChange}
        />
      </div>
      <div>
        <label>Quantidade de avaliações:</label>
        <input
          type="number"
          name="quantidade"
          value={formData.avaliacao.quantidade}
          onChange={handleAvaliacaoChange}
        />
      </div>
      <button type="submit">Salvar</button>
    </form>
  );
};
